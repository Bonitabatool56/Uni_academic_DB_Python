-- Department Table
CREATE TABLE Department (
  DeptID INT PRIMARY KEY,
  DeptName VARCHAR(50),
  HOD VARCHAR(50),
  Email VARCHAR(50),
  Phone VARCHAR(20)
);

-- Teacher Table
CREATE TABLE Teacher (
  TeacherID INT PRIMARY KEY,
  Name VARCHAR(50),
  Email VARCHAR(50),
  Phone VARCHAR(20),
  DeptID INT,
  FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
);

-- Course Table
CREATE TABLE Course (
  CourseID INT PRIMARY KEY,
  CourseName VARCHAR(50),
  CreditHours INT,
  DeptID INT,
  TeacherID INT,
  FOREIGN KEY (DeptID) REFERENCES Department(DeptID),
  FOREIGN KEY (TeacherID) REFERENCES Teacher(TeacherID)
);

-- Student Table
CREATE TABLE Student (
  StudentID INT PRIMARY KEY,
  Name VARCHAR(50),
  Email VARCHAR(50),
  Phone VARCHAR(20),
  BatchID INT,
  DeptID INT,
  FOREIGN KEY (BatchID) REFERENCES Batch(BatchID),
  FOREIGN KEY (DeptID) REFERENCES Department(DeptID)
);

-- ✅ FIX: make sure Enrollment StudentID references valid StudentID
CREATE TABLE Enrollment (
  EnrollID INT PRIMARY KEY,
  StudentID INT,
  CourseID INT,
  EnrollDate DATE,
  Status VARCHAR(20),
  FOREIGN KEY (StudentID) REFERENCES Student(StudentID)
      ON UPDATE CASCADE ON DELETE CASCADE,
  FOREIGN KEY (CourseID) REFERENCES Course(CourseID)
      ON UPDATE CASCADE ON DELETE CASCADE
);

-- Examination Table
CREATE TABLE Examination (
  ExamID INT PRIMARY KEY,

  ExamDate DATE,
  Type VARCHAR(30),
  Venue VARCHAR(50),
  FOREIGN KEY (CourseID) REFERENCES Course(CourseID)

-- ======================= SAMPLE QUERIES =======================

-- 1. List all students with their department and enrolled courses
SELECT 
  s.Name AS StudentName,
  d.DeptName,
  c.CourseName
FROM Student s
JOIN Department d ON s.DeptID = d.DeptID
JOIN Enrollment e ON s.StudentID = e.StudentID
JOIN Course c ON e.CourseID = c.CourseID;

-- 2. List teachers with courses they teach
SELECT 
  t.Name AS TeacherName,
  c.CourseName
FROM Teacher t
JOIN Course c ON t.TeacherID = c.TeacherID;

-- 3. List all active enrollments
SELECT 
  s.Name AS StudentName,
  c.CourseName,
  e.EnrollDate,
  e.Status
FROM Enrollment e
JOIN Student s ON e.StudentID = s.StudentID
JOIN Course c ON e.CourseID = c.CourseID
WHERE e.Status = 'Active';

-- 4. Show student results with grade and exam info
SELECT 
  s.Name AS StudentName,
  c.CourseName,
  g.GradeLetter,
  g.GPA,
  e.Type AS ExamType,
  e.ExamDate
FROM Result r
JOIN Student s ON r.StudentID = s.StudentID
JOIN Course c ON r.CourseID = c.CourseID
JOIN Grade g ON r.GradeID = g.GradeID
JOIN Examination e ON r.ExamID = e.ExamID;
